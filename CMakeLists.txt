cmake_minimum_required(VERSION 3.24)
project(LissQtLabs VERSION 1.0.0 LANGUAGES CXX)
set(CMAKE_CXX_STANDARD 20)
set(CMAKE_CXX_STANDARD_REQUIRED ON)
set(CMAKE_AUTOMOC ON)
find_package(Qt6 6.8 REQUIRED COMPONENTS Core Gui Widgets Concurrent)
add_compile_definitions(QT_NO_CAST_FROM_ASCII QT_NO_CAST_TO_ASCII)
if(MSVC)
  add_compile_options(/utf-8 /W4 /permissive- /EHsc)
endif()
add_library(labs_core STATIC src/os_core.cpp src/huffman_core.cpp src/calc_core.cpp)
target_include_directories(labs_core PUBLIC src)
target_link_libraries(labs_core PUBLIC Qt6::Core)
add_library(labs_ui STATIC src/common.cpp src/os_window.cpp src/huffman_window.cpp src/calc_window.cpp)
target_link_libraries(labs_ui PUBLIC labs_core Qt6::Gui Qt6::Widgets Qt6::Concurrent)
foreach(app IN ITEMS OsLab HuffmanLab Calculator)
  qt_add_executable(Liss${app} WIN32 src/main.cpp)
  if(WIN32)
    target_sources(Liss${app} PRIVATE src/app.rc)
  endif()
  target_compile_definitions(Liss${app} PRIVATE LAB_APP_${app})
  target_link_libraries(Liss${app} PRIVATE labs_ui)
endforeach()
add_executable(QtLabsChecks tests/checks.cpp)
target_link_libraries(QtLabsChecks PRIVATE labs_core)
enable_testing()
add_test(NAME core_acceptance COMMAND QtLabsChecks)
